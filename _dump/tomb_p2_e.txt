465:       }
466:       return;
467:     }
468:     if (!key || seen.has(key)) return;
469:     if (isExperimentDeleted(t, records)) return;
470:     seen.add(key);
471:     added.push(t);
472:     fresh.push(t);
473:   });
474:   const tests = withoutDeletedExperiments(fresh, records);
475:   return { tests, added, aliases, collapsed };
476: };
