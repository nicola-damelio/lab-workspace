575: /** LE GESTE : tout ce qui a disparu entre `before` et `after` par une ACTION de
576:  *  l'utilisateur (suppression 🗑, undo/redo…) laisse sa suppression enregistrée
577:  *  — c'est ce qui distingue une suppression d'une perte. Rend la liste à jour
578:  *  des suppressions. */
579: export const recordExperimentDeletions = (before, after, datasetArg, deletedAt = Date.now()) => {
580:   const still = new Set(
581:     (Array.isArray(after) ? after : []).filter(Boolean).map(experimentKey).filter(Boolean)
582:   );
583:   const gone = (Array.isArray(before) ? before : []).filter((t) => {
584:     const key = experimentKey(t);
585:     return !!key && !still.has(key);
586:   });
587:   /* Rien n'a disparu (le cas de très loin le plus fréquent : une frappe dans une
588:      page d'expérience) : on ne réécrit même pas la clé. */
589:   if (!gone.length) return loadExperimentDeletions();
590:   const scope = datasetArg === undefined || datasetArg === null ? '' : String(datasetArg);
591:   let list = loadExperimentDeletions();
592:   gone.forEach((t) => { list = addDeletion(list, { ...t, datasetId: scope || datasetOf(t) }, deletedAt); });
593:   return saveExperimentDeletions(list);
594: };
