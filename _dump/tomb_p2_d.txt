358: /** CE QUI ENTRE À L'ÉCRAN QUAND ON APPLIQUE UNE COPIE (➕ « Add the selected

361:  *  empilée.

364:  *  ramène l'expérience sous son nom d'aujourd'hui avec l'id qu'elle a toujours
365:  *  eu ; l'anti-jumeau qui ne comparait que des NOMS (l'écran avait sa règle, le
366:  *  verrou la sienne) en empilait une SECONDE ligne, et la relecture suivante —
367:  *  elle, ID-keyée — effaçait l'une des deux sans un mot.

370:  *  même essai portent le même nom et des ids différents, et c'est l'essai ENTIER
371:  *  qui doit revenir. Le nom ne sépare que de ce que le dataset PORTE DÉJÀ.

374:  *  entrée sans id en reçoit un, sinon rien ne pourrait plus la reconnaître) et
375:  *  celles écartées — `{ test, twin, by }`, `by` valant `'id'` ou `'name'` — pour
376:  *  que l'appelant le DISE (voir `describeSkippedAdditions`).

378: export const experimentsToAdd = (current, incoming, makeId) => {
379:   const screen = (Array.isArray(current) ? current : []).filter(Boolean);
380:   const ids = new Set(screen.map(idOf).filter(Boolean));
381:   const names = new Set(screen.map(nameOf).filter(Boolean));
382:   const newId = typeof makeId === 'function' ? makeId : defaultFreshId;
383:   const tests = [];
384:   const skipped = [];
385:   (Array.isArray(incoming) ? incoming : []).forEach((raw) => {
386:     const row = raw && typeof raw === 'object' ? raw : { name: String(raw === undefined || raw === null ? '' : raw) };
387:     const id = idOf(row);
388:     const name = nameOf(row);
389:     const twin = id && ids.has(id)
390:       ? screen.find((t) => idOf(t) === id)
391:       : (name && names.has(name) ? screen.find((t) => nameOf(t) === name) : null);
392:     if (twin) { skipped.push({ test: row, twin, by: id && idOf(twin) === id ? 'id' : 'name' }); return; }
393:     if (id) ids.add(id);
394:     if (name) names.add(name);
395:     tests.push(id ? { ...row } : { ...row, id: newId() });
396:   });
397:   return { tests, skipped };
