130:     .sort((a, b) => b[field] - a[field])
131:     .slice(0, MAX_RECORDS);

137: /** Enregistrer la suppression d'une expérience (la plus récente en tête). */
138: export const addDeletion = (deletions, test, deletedAt = Date.now()) => {

141:     name: nameOf(test),
142:     datasetId: datasetOf(test),
143:     deletedAt: Number(deletedAt) || Date.now()
144:   };

146:   return normalizeDeletions([entry, ...(Array.isArray(deletions) ? deletions : [])]);

170: /** La RÈGLE d'identité d'un record, une seule fois : même id (ou même nom quand
171:  *  l'un des deux n'a pas d'id), et un dataset absent d'un côté au moins (ou
172:  *  identique) — un record sans dataset vaut partout. */
173: const sameStamped = (a, b) => {
174:   if (!a || !b) return false;
175:   const aId = idOf(a);
176:   const bId = idOf(b);
177:   const sameName = !!nameOf(a) && nameOf(a) === nameOf(b);
178:   if (!(aId && bId ? aId === bId : sameName)) return false;
179:   return !a.datasetId || !b.datasetId || a.datasetId === b.datasetId;
180: };

182: /** LES SUPPRESSIONS QUI TIENNENT ENCORE : celles qu'aucune levée plus récente
183:  *  (ou de même date) n'a annulées. C'est la seule lecture des suppressions
184:  *  d'expériences — lever une suppression est donc vu de partout d'un coup. */
185: export const withoutRevivedExperiments = (deletions, revivals) => {
186:   const lifted = normalizeRevivals(revivals);
187:   const records = normalizeDeletions(deletions);
188:   if (!lifted.length) return records;
189:   return records.filter((d) => !lifted.some(
190:     (r) => sameStamped(r, d) && (Number(r.revivedAt) || 0) >= (Number(d.deletedAt) || 0)
191:   ));
192: };
