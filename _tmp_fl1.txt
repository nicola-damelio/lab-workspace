 *  quand le libellÃ© d'une entrÃ©e vient du NOM DU FICHIER sur le Drive
 *  (`labelFromDriveFileName`, une bibliothÃ¨que reconstruite par une lecture), il
 *  porte DÃ‰JÃ€ l'empreinte d'hier ; l'Ã©criture suivante la rajoutait Ã  la fin,
 *  puis la lecture suivante relisait ce nom-lÃ , et ainsi de suite :

 *      Fig2-1b5tpha-1b5tpha-1b5tpha-1b5tpha-1b5tpha.jpg.meta.json
 *
 *  Une empreinte dÃ©jÃ  prÃ©sente Ã  la FIN du libellÃ© n'est donc jamais rÃ©pÃ©tÃ©e, et
 *  un nom dÃ©jÃ  abÃ®mÃ© est RAMENÃ‰ au bon dÃ¨s la prochaine Ã©criture (l'empreinte est

export const figureFileName = (label, ext, identity = '') => {
  const tag = String(identity || '').trim() ? `-${fileTagOf(identity)}` : '';
  let base = sanitizeSlug(label) || 'figure';
  while (tag && base.endsWith(tag)) base = base.slice(0, -tag.length);
  return `${base || 'figure'}${tag}.${ext}`;
};

/** LE dossier d'images Ã  utiliser pour ce projet â€” celui qui EXISTE.
 *
 *  `create:false` (toutes les LECTURES : relire la bibliothÃ¨que, retrouver un
 *  sidecar) ne crÃ©e jamais rien : quand aucun dossier n'est identifiÃ©, on
 *  retourne `leafId:''` et l'appelant dit ce qu'il a vu (`candidates`) au lieu de
 *  fabriquer un dossier vide Ã  cÃ´tÃ© des fichiers.
 *
 *  `create:true` (les Ã‰CRITURES : publier une figure, dÃ©poser son sidecar)
 *  rejoint le dossier existant â€” sous son ancien nom s'il a Ã©tÃ© renommÃ© â€” et ne
 *  crÃ©e l'emplacement canonique que s'il n'y a vraiment rien.
 *
 *  @returns {Promise<{ name:string, leafId:string, exact:boolean, via:string,
 *                      folder:string, candidates:Array }>} */
const figuresFolderFor = async (projectName, { create = false } = {}) => {
  /* Le chemin annoncÃ© est celui qui EXISTE sur le Drive (seau commun : Â« unassigned Â»,
     `unassigned`, voir imagesFolderPathOnDrive) : l'Ã©cran dit alors exactement ce
     que le Drive montre. */
  const wanted = imagesFolderPathOnDrive(projectName);
  const found = await findProjectFiguresFolder(projectName).catch(() => null);
  if (found && found.leafId) return found;
  if (!create) {
    return {
      name: '', leafId: '', exact: false, via: '', folder: wanted,
      candidates: (found && found.candidates) || []
    };
  }
  const resolved = await resolveDrivePathFromNames(projectImagesFolderPath(projectName)).catch(() => null);
  /* La rÃ©solution rend les noms RÃ‰ELS de chaque segment : `projects`, puis le nom
     du dossier de projet (celui du Drive â€” le seau `unassigned` quand il n'y a pas
     de projet), puis `images`. */
  const realPath = (resolved && Array.isArray(resolved.path) ? resolved.path : [])
    .map((p) => String((p && p.name) || '')).filter(Boolean);
  const name = realPath.length > 1
    ? realPath[1]
    : (sanitizeSlug(projectName) || unassignedFolderNames()[0]);
  return {
    name, leafId: (resolved && resolved.leafId) || '', exact: true, via: 'created',
    folder: realPath.length ? realPath.join('/') : wanted, candidates: []
  };
};
